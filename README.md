Визначити значення заданої за варіантом кусочно-безперервної функції $y(x)$, якщо воно існує, або вивести на екран повідомлення про неіснування функції для заданого $x$.Функція задана формулою:
$$y = \begin{cases} 10x^3 + \frac{7x}{5} + 2, & x \in [-49, -10) \cup (0, 10] \\  -x + 9, & x \in (20, +\infty) \end{cases}$$Задачу необхідно розв'язати двома способами (написати дві окремі програми):Програма 1: Використовуються лише одиничні операції порівняння ($=, \neq, <, \le, >, \ge$) без застосування логічних операцій (not, and, or).Програма 2: Обов'язково використовуються булеві (логічні) операції без надмірності.💻 Програма 1 (Без логічних операцій)Текст програми#include <stdio.h>

int main(void)
{
    double x, y;
    printf("Enter x: ");
    scanf("%lf", &x);
    
    if (x >= -49)
    {
        if (x < -10)
        {
            y = 10 * x * x * x + (7 * x) / 5 + 2;
            printf("y = %lf", y);
        }
        else
        {
            if (x <= 0)
            {
                printf("Error: y(x) doesn't exist");
            }
            else
            {
                if (x <= 10)
                {
                    y = 10 * x * x * x + (7 * x) / 5 + 2;
                    printf("y = %lf", y);
                }
                else
                {
                    if (x <= 20)
                    {
                        printf("Error: y(x) doesn't exist");
                    }
                    else
                    {
                        y = -x + 9;
                        printf("y = %lf", y);
                    }
                }
            }
        }
    }
    else
    {
        printf("Error: y(x) doesn't exist");
    }
    return 0;
}
💻 Програма 2 (З використанням булевих операцій)Текст програми#include <stdio.h>

int main(void)
{
    double x, y;
    printf("Enter x: ");
    scanf("%lf", &x);
    
    if (x >= -49 && x < -10 || x > 0 && x <= 10)
    {
        y = 10 * x * x * x + (7 * x) / 5 + 2;
        printf("y = %lf", y);
    } 
    else 
    {
        if (x > 20)
        {
            y = -x + 9;
            printf("y = %lf", y);
        }
        else
        {
            printf("Error: y(x) doesn't exist");
        }
    }
    return 0;
}
